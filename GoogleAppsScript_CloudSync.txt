const PROP_KEY = 'PICKLEBALL_2COURTS_DATA';

function doGet(e) {
  const props = PropertiesService.getScriptProperties();
  const raw = props.getProperty(PROP_KEY);
  const data = raw ? JSON.parse(raw) : { updatedAt: Date.now(), bookings: {} };
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const props = PropertiesService.getScriptProperties();
  const body = JSON.parse(e.postData.contents || '{}');
  const payload = {
    updatedAt: Date.now(),
    bookings: body.bookings || {}
  };
  props.setProperty(PROP_KEY, JSON.stringify(payload));
  return ContentService.createTextOutput(JSON.stringify({ ok: true, updatedAt: payload.updatedAt }))
    .setMimeType(ContentService.MimeType.JSON);
}
