const PROP_KEY = 'PICKLEBALL_2COURTS_DATA';

function doGet(e) {
  const props = PropertiesService.getScriptProperties();
  const raw = props.getProperty(PROP_KEY);
  const data = raw ? JSON.parse(raw) : { updatedAt: Date.now(), bookings: {} };
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const props = PropertiesService.getScriptProperties();
    const raw = props.getProperty(PROP_KEY);
    const current = raw ? JSON.parse(raw) : { updatedAt: Date.now(), bookings: {} };
    const incoming = JSON.parse(e.postData.contents || '{}');

    let mergedBookings = Object.assign({}, current.bookings || {});
    if (incoming.replaceAll) {
      mergedBookings = incoming.bookings || {};
    } else if (incoming.delta) {
      if (incoming.delta.set) {
        Object.assign(mergedBookings, incoming.delta.set);
      }
      if (Array.isArray(incoming.delta.delete)) {
        incoming.delta.delete.forEach(function(k) { delete mergedBookings[k]; });
      }
    } else if (incoming.bookings) {
      Object.assign(mergedBookings, incoming.bookings);
    }

    const payload = {
      updatedAt: Date.now(),
      bookings: mergedBookings
    };
    props.setProperty(PROP_KEY, JSON.stringify(payload));
    return ContentService.createTextOutput(JSON.stringify({
      ok: true,
      updatedAt: payload.updatedAt,
      bookings: mergedBookings
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
